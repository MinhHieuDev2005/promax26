cmake_minimum_required(VERSION 3.20)
project(udon_chokudai LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Build the unchanged core in a separate directory. No edits to its dispatcher.
add_subdirectory(../.. legacy EXCLUDE_FROM_ALL)
add_library(chokudai_search solver.cpp)
target_include_directories(chokudai_search PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(chokudai_search PUBLIC hexudon_core)
add_executable(chokudai_runner main.cpp)
target_link_libraries(chokudai_runner PRIVATE chokudai_search)
add_executable(chokudai_tests tests.cpp)
target_link_libraries(chokudai_tests PRIVATE chokudai_search)
add_executable(chokudai_benchmark benchmark.cpp)
target_link_libraries(chokudai_benchmark PRIVATE chokudai_search)
if(MINGW)
  # A standalone runner can be selected in the existing GUI without copying
  # compiler DLLs or replacing any DLL shipped with the old solvers.
  foreach(target chokudai_runner chokudai_tests chokudai_benchmark)
    target_link_options(${target} PRIVATE -static)
  endforeach()
endif()
enable_testing()
add_test(NAME chokudai_tests COMMAND chokudai_tests)
