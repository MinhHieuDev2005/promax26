cmake_minimum_required(VERSION 3.20)
project(hexudon_horizon_portfolio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_subdirectory(../.. legacy EXCLUDE_FROM_ALL)
add_executable(horizon_portfolio_runner main.cpp)
target_link_libraries(horizon_portfolio_runner PRIVATE hexudon_core)
add_executable(day_portfolio_runner main.cpp)
target_compile_definitions(day_portfolio_runner PRIVATE DAY_PORTFOLIO=1)
target_link_libraries(day_portfolio_runner PRIVATE hexudon_core)
add_executable(reachability_audit reachability_audit.cpp)
target_link_libraries(reachability_audit PRIVATE hexudon_core)
add_executable(subset_portfolio_runner main.cpp)
target_compile_definitions(subset_portfolio_runner PRIVATE DAY_PORTFOLIO=1 SUBSET_REPAIR=1)
target_link_libraries(subset_portfolio_runner PRIVATE hexudon_core)
if(MINGW)
  target_link_options(horizon_portfolio_runner PRIVATE -static)
  target_link_options(day_portfolio_runner PRIVATE -static)
  target_link_options(subset_portfolio_runner PRIVATE -static)
endif()
