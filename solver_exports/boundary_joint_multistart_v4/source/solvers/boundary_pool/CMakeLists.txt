cmake_minimum_required(VERSION 3.20)
project(hexudon_boundary_pool LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_subdirectory(../.. legacy EXCLUDE_FROM_ALL)
add_executable(boundary_pool_runner main.cpp)
target_link_libraries(boundary_pool_runner PRIVATE hexudon_core)
add_executable(boundary_pool_preview_runner main.cpp)
target_compile_definitions(boundary_pool_preview_runner PRIVATE BOUNDARY_DEFAULT_PREVIEW=1)
target_link_libraries(boundary_pool_preview_runner PRIVATE hexudon_core)
add_executable(boundary_pool_tests tests.cpp)
target_link_libraries(boundary_pool_tests PRIVATE hexudon_core)
if(MINGW)
  target_link_options(boundary_pool_runner PRIVATE -static)
  target_link_options(boundary_pool_preview_runner PRIVATE -static)
  target_link_options(boundary_pool_tests PRIVATE -static)
endif()
