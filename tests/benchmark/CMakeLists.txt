cmake_minimum_required(VERSION 3.20)
project(hexa_udon_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
if(NOT SOLVER_SOURCE_ROOT)
    get_filename_component(SOLVER_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_subdirectory("${SOLVER_SOURCE_ROOT}" solver)
add_executable(solver_bench match.cpp)
target_link_libraries(solver_bench PRIVATE hexa_udon_prematch_selector hexa_udon_json)
add_executable(solver_bench_checks EXCLUDE_FROM_ALL "${SOLVER_SOURCE_ROOT}/tests/unit/app_test.cpp")
target_link_libraries(solver_bench_checks PRIVATE hexa_udon_app)
